CREATE DATABASE RSGL;
USE RSGL;

CREATE TABLE Departments (
    D_ID TINYINT NOT NULL PRIMARY KEY,
    D_name VARCHAR(10) NOT NULL
);

CREATE TABLE Employees (
    E_ID tinyint NOT NULL PRIMARY KEY,
    E_name VARCHAR(8),
    sex VARCHAR(2) DEFAULT '男',
    professional VARCHAR(6),
    political VARCHAR(8),
    education VARCHAR(8),
    birth DATE,
    marry VARCHAR(8),
    GZ_time DATE,
    D_ID VARCHAR(5),
    BZ VARCHAR(2) DEFAULT '是'
);

CREATE TABLE Salary (
    E_ID TINYINT NOT NULL,
    month DATE,
    JIB_IN FLOAT(6,2),
    JIX_IN FLOAT(6,2),
    JINT_IN FLOAT(6,2),
    GJ_OUT FLOAT(6,2),
    TAX_OUT FLOAT(6,2),
    QT_OUT FLOAT(6,2)
);

# 外键
ALTER TABLE Employees
ADD CONSTRAINT fk_employees_departments
FOREIGN KEY (E_ID) REFERENCES Departments(D_ID)
ON UPDATE CASCADE
ON DELETE CASCADE;

ALTER TABLE Salary
ADD CONSTRAINT fk_salary_employees
FOREIGN KEY (E_ID) REFERENCES Employees(E_ID)
ON UPDATE CASCADE
ON DELETE CASCADE;

-- D_name 唯一约束
ALTER TABLE Departments
ADD CONSTRAINT unique_dname UNIQUE (D_name);
